# Project 6: Student Grade Management System

## 1. Real-World Scenario & Theoretical Foundation

Educational institutions and universities require robust software to manage student profiles, record academic grades across multiple subjects, compute Grade Point Averages (GPAs), and maintain institutional records. Each student is modeled as an instance of the `Student` class, encapsulating their personal metadata and dynamic grade dictionary within a centralized academic database.

---

## 2. Core OOP Concepts & Pillars Used

* **Encapsulation:** Grouping student details (`name`, `student_id`) and course performance dictionary (`grades`) together inside the `Student` class alongside specialized behaviors (`add_grade`, `update_grade`, `calculate_gpa`).
* **Dictionary State Mapping:** Utilizing key-value pairs (`{subject_name: score}`) to store, retrieve, and update individual subject scores efficiently.
* **CRUD Architecture:**
* **Create:** Enroll a new student into the academic database.
* **Read:** View student profile, individual subject grades, and cumulative GPA.
* **Update:** Add new grades, update existing scores, or modify student metadata.
* **Delete:** Remove/unenroll student records from the database.



---

## 3. Complete Python Source Code (CLI & Core Logic with Full CRUD)

```python
class Student:
    def __init__(self, name, student_id):
        self.name = name
        self.student_id = str(student_id)
        self.grades = {}  # Format: {subject_name: numeric_score}

    def add_or_update_grade(self, subject, score):
        try:
            val = float(score)
            if 0 <= val <= 100:
                self.grades[subject] = val
                print(f"Success: Grade recorded for '{subject}' ({val}) for student {self.name}.")
            else:
                print("Error: Score must be between 0 and 100.")
        except ValueError:
            print("Error: Score must be a valid number.")

    def remove_grade(self, subject):
        if subject in self.grades:
            del self.grades[subject]
            print(f"Success: Subject '{subject}' removed from record.")
        else:
            print(f"Error: Subject '{subject}' not found in student records.")

    def calculate_gpa(self):
        if not self.grades:
            return 0.0
        total_points = sum(self.grades.values())
        return total_points / len(self.grades)

    def display_student_record(self):
        print("\n" + "="*42)
        print("          STUDENT ACADEMIC REPORT")
        print("="*42)
        print(f"Student Name : {self.name}")
        print(f"Student ID   : {self.student_id}")
        print("Grades Record:")
        if not self.grades:
            print("  No grades recorded yet.")
        else:
            for sub, score in self.grades.items():
                print(f"  - {sub:<15} : {score:>5.1f}")
        print(f"Cumulative GPA Score: {self.calculate_gpa():.2f}%")
        print("="*42)

def find_student(database, student_id):
    for student in database:
        if student.student_id == student_id:
            return student
    return None

def main():
    database = []

    while True:
        print("\n--- STUDENT GRADE MANAGEMENT SYSTEM (CRUD) ---")
        print("1. Enroll New Student (Create)")
        print("2. View Student Report (Read)")
        print("3. Add/Update Subject Grade (Update)")
        print("4. Remove Subject Grade (Delete)")
        print("5. Delete Student Record (Delete)")
        print("6. Exit")

        choice = input("Enter your choice (1-6): ").strip()

        if choice == '1':
            print("\n--- Enroll New Student ---")
            s_id = input("Enter Student ID: ").strip()
            if find_student(database, s_id):
                print("Error: Student ID already exists!")
                continue
            name = input("Enter Student Name: ").strip()
            
            database.append(Student(name, s_id))
            print(f"Success! Enrolled student {name}.")

        elif choice == '2':
            print("\n--- View Student Report ---")
            s_id = input("Enter Student ID: ").strip()
            student = find_student(database, s_id)
            if student:
                student.display_student_record()
            else:
                print("Error: Student ID not found.")

        elif choice == '3':
            print("\n--- Add/Update Grade ---")
            s_id = input("Enter Student ID: ").strip()
            student = find_student(database, s_id)
            if student:
                sub = input("Enter Subject Name: ").strip()
                score = input("Enter Numeric Score (0-100): ").strip()
                student.add_or_update_grade(sub, score)
            else:
                print("Error: Student ID not found.")

        elif choice == '4':
            print("\n--- Remove Subject Grade ---")
            s_id = input("Enter Student ID: ").strip()
            student = find_student(database, s_id)
            if student:
                sub = input("Enter Subject Name to remove: ").strip()
                student.remove_grade(sub)
            else:
                print("Error: Student ID not found.")

        elif choice == '5':
            print("\n--- Delete Student Record ---")
            s_id = input("Enter Student ID to delete: ").strip()
            student = find_student(database, s_id)
            if student:
                database.remove(student)
                print(f"Student ID {s_id} removed from database.")
            else:
                print("Error: Student ID not found.")

        elif choice == '6':
            print("Exiting Student Grade Management System. Goodbye!")
            break
        else:
            print("Invalid option. Please choose between 1 and 6.")

main()
```
