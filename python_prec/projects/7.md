# Project 7: Restaurant Table Reservation System

## 1. Real-World Scenario & Theoretical Foundation

Restaurant management software requires real-time coordination of dining room layouts, table capacities, and customer seating reservations. Host staff need to track whether a table is available or booked, assign guest names to specific tables, update party sizes, and manage floor inventories dynamically. Each dining table is modeled as an instance of the `RestaurantTable` class, encapsulating its attributes and booking status.

---

## 2. Core OOP Concepts & Pillars Used

* **Encapsulation:** Bundling table attributes (`table_number`, `capacity`, `is_reserved`, `guest_name`) and booking behaviors (`reserve_table`, `cancel_reservation`, `update_table`) inside the `RestaurantTable` class.
* **Boolean State Management:** Utilizing `is_reserved` flags to prevent double-booking conflicts and manage seating availability instantly.
* **CRUD Architecture:**
* **Create:** Add new dining tables to the restaurant floor plan inventory.
* **Read:** View table statuses, capacities, and assigned guest details.
* **Update:** Reserve a table for a guest, update table seating capacity, or cancel a booking.
* **Delete:** Remove tables from the floor layout catalog.



---

## 3. Complete Python Source Code (CLI & Core Logic with Full CRUD)

```python
class RestaurantTable:
    def __init__(self, table_number, capacity):
        self.table_number = str(table_number)
        self.capacity = int(capacity)
        self.is_reserved = False
        self.guest_name = None

    def reserve_table(self, guest_name, party_size):
        if not self.is_reserved:
            if int(party_size) <= self.capacity:
                self.is_reserved = True
                self.guest_name = guest_name
                print(f"Success: Table {self.table_number} reserved for {guest_name} (Party of {party_size}).")
            else:
                print(f"Error: Party size ({party_size}) exceeds table capacity ({self.capacity}).")
        else:
            print(f"Error: Table {self.table_number} is already reserved by {self.guest_name}!")

    def cancel_reservation(self):
        if self.is_reserved:
            print(f"Success: Reservation for Table {self.table_number} ({self.guest_name}) cancelled.")
            self.is_reserved = False
            self.guest_name = None
        else:
            print(f"Error: Table {self.table_number} is currently not reserved.")

    def update_table(self, new_capacity):
        self.capacity = int(new_capacity)
        print(f"Success: Table {self.table_number} capacity updated to {new_capacity} seats.")

    def display_status(self):
        status = f"Reserved for {self.guest_name}" if self.is_reserved else "Available"
        print(f"Table #{self.table_number} | Capacity: {self.capacity} seats | Status: {status}")

def find_table(floor_plan, t_num):
    for table in floor_plan:
        if table.table_number == str(t_num):
            return table
    return None

def main():
    floor_plan = [
        RestaurantTable(1, 2),
        RestaurantTable(2, 4),
        RestaurantTable(3, 6),
        RestaurantTable(4, 8)
    ]

    while True:
        print("\n--- RESTAURANT TABLE RESERVATION SYSTEM (CRUD) ---")
        print("1. Add New Table to Floor Plan (Create)")
        print("2. View Floor Status (Read)")
        print("3. Reserve Table for Guest (Update)")
        print("4. Cancel Table Reservation (Update)")
        print("5. Delete Table from Floor (Delete)")
        print("6. Exit")

        choice = input("Enter your choice (1-6): ").strip()

        if choice == '1':
            print("\n--- Add New Table ---")
            t_num = input("Enter Table Number: ").strip()
            if find_table(floor_plan, t_num):
                print("Error: Table number already exists!")
                continue
            try:
                cap = int(input("Enter Seating Capacity: "))
                floor_plan.append(RestaurantTable(t_num, cap))
                print(f"Success! Table {t_num} added to floor plan.")
            except ValueError:
                print("Invalid capacity format.")

        elif choice == '2':
            print("\n--- Restaurant Floor Status ---")
            for table in floor_plan:
                table.display_status()

        elif choice == '3':
            print("\n--- Reserve Table ---")
            t_num = input("Enter Table Number: ").strip()
            table = find_table(floor_plan, t_num)
            if table:
                guest = input("Enter Guest Name: ").strip()
                try:
                    party = int(input("Enter Party Size: "))
                    table.reserve_table(guest, party)
                except ValueError:
                    print("Invalid party size number format.")
            else:
                print("Error: Table not found.")

        elif choice == '4':
            print("\n--- Cancel Reservation ---")
            t_num = input("Enter Table Number to cancel: ").strip()
            table = find_table(floor_plan, t_num)
            if table:
                table.cancel_reservation()
            else:
                print("Error: Table not found.")

        elif choice == '5':
            print("\n--- Delete Table ---")
            t_num = input("Enter Table Number to delete: ").strip()
            table = find_table(floor_plan, t_num)
            if table:
                if table.is_reserved:
                    print("Error: Cannot delete a table with an active reservation!")
                else:
                    floor_plan.remove(table)
                    print(f"Table {t_num} removed from floor plan.")
            else:
                print("Error: Table not found.")

        elif choice == '6':
            print("Exiting Restaurant Reservation System. Goodbye!")
            break
        else:
            print("Invalid option. Please choose between 1 and 6.")

main()
```
