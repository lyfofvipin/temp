# Project 8: Movie Ticket Booking System

## 1. Real-World Scenario & Theoretical Foundation

Cinema theater software manages movie showtimes, seat availability, ticket sales, and customer bookings. Box office systems must track individual ticket purchases, verify seat availability to prevent double-booking, compute pricing based on ticket tiers, and allow ticket modifications or cancellations. Each ticket or booking is modeled as an instance of the `MovieTicket` class.

---

## 2. Core OOP Concepts & Pillars Used

* **Encapsulation:** Bundling ticket attributes (`booking_id`, `movie_title`, `seat_number`, `patron_name`, `price`) and operational methods (`update_seat`, `cancel_ticket`) inside the `MovieTicket` class.
* **State Management:** Tracking active booking states and seat inventory to prevent scheduling conflicts.
* **CRUD Architecture:**
* **Create:** Book a new movie ticket with seat selection and patron details.
* **Read:** View active booking details, movie schedules, and ticket statuses.
* **Update:** Modify seat assignments or update patron contact profiles.
* **Delete:** Cancel/delete movie ticket bookings and release seats back to the inventory.



---

## 3. Complete Python Source Code (CLI & Core Logic with Full CRUD)

```python
class MovieTicket:
    def __init__(self, booking_id, movie_title, seat_number, patron_name, price):
        self.booking_id = str(booking_id)
        self.movie_title = movie_title
        self.seat_number = str(seat_number)
        self.patron_name = patron_name
        self.price = float(price)

    def update_seat(self, new_seat):
        self.seat_number = str(new_seat)
        print(f"Success: Booking {self.booking_id} seat updated to {new_seat}.")

    def update_patron(self, new_name):
        self.patron_name = new_name
        print(f"Success: Patron name for booking {self.booking_id} updated to {new_name}.")

    def display_ticket_info(self):
        print("\n" + "="*42)
        print("          CINEMA TICKET STATEMENT")
        print("="*42)
        print(f"Booking ID   : {self.booking_id}")
        print(f"Movie Title  : {self.movie_title}")
        print(f"Seat Number  : {self.seat_number}")
        print(f"Patron Name  : {self.patron_name}")
        print(f"Ticket Price : ${self.price:.2f}")
        print("="*42)

def find_booking(bookings_list, b_id):
    for ticket in bookings_list:
        if ticket.booking_id == str(b_id):
            return ticket
    return None

def main():
    bookings = [
        MovieTicket("B101", "Interstellar", "A-12", "Alice Smith", 14.50),
        MovieTicket("B102", "Inception", "B-05", "Bob Jones", 12.00)
    ]

    while True:
        print("\n--- MOVIE TICKET BOOKING SYSTEM (CRUD) ---")
        print("1. Book New Movie Ticket (Create)")
        print("2. View Ticket Details (Read)")
        print("3. Update Seat / Patron Name (Update)")
        print("4. Cancel/Delete Ticket Booking (Delete)")
        print("5. Exit")

        choice = input("Enter your choice (1-5): ").strip()

        if choice == '1':
            print("\n--- Book New Ticket ---")
            b_id = input("Enter Booking ID: ").strip()
            if find_booking(bookings, b_id):
                print("Error: Booking ID already exists!")
                continue
            title = input("Enter Movie Title: ").strip()
            seat = input("Enter Seat Number (e.g., C-04): ").strip()
            patron = input("Enter Patron Name: ").strip()
            try:
                price = float(input("Enter Ticket Price ($): "))
                bookings.append(MovieTicket(b_id, title, seat, patron, price))
                print(f"Success! Ticket booked for {patron} for '{title}'.")
            except ValueError:
                print("Invalid price number format.")

        elif choice == '2':
            print("\n--- View Ticket Details ---")
            b_id = input("Enter Booking ID to search: ").strip()
            ticket = find_booking(bookings, b_id)
            if ticket:
                ticket.display_ticket_info()
            else:
                print("Error: Booking ID not found.")

        elif choice == '3':
            print("\n--- Update Booking Details ---")
            b_id = input("Enter Booking ID to update: ").strip()
            ticket = find_booking(bookings, b_id)
            if ticket:
                print("1. Update Seat Number")
                print("2. Update Patron Name")
                sub = input("Choose update type (1/2): ").strip()
                if sub == '1':
                    new_seat = input("Enter new seat number: ").strip()
                    ticket.update_seat(new_seat)
                elif sub == '2':
                    new_name = input("Enter new patron name: ").strip()
                    ticket.update_patron(new_name)
                else:
                    print("Invalid update option.")
            else:
                print("Error: Booking ID not found.")

        elif choice == '4':
            print("\n--- Cancel/Delete Ticket ---")
            b_id = input("Enter Booking ID to cancel: ").strip()
            ticket = find_booking(bookings, b_id)
            if ticket:
                bookings.remove(ticket)
                print(f"Booking ID {b_id} has been cancelled and deleted.")
            else:
                print("Error: Booking ID not found.")

        elif choice == '5':
            print("Exiting Movie Ticket System. Goodbye!")
            break
        else:
            print("Invalid option. Please choose between 1 and 5.")

main()

```
