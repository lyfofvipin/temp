# Project 9: Smart Shopping Cart & Invoice System (File Handling & Dunder Methods)

## 1. Real-World Scenario & Theoretical Foundation

E-commerce platforms rely on persistent shopping carts that save user items even if they close the browser. This system models a smart shopping cart where customers can add items, check out, persist their cart data locally using file handling (`json`), and use **operator overloading (dunder methods)** to combine two carts together (e.g., merging a wish list or sharing a cart with a partner) or remove items seamlessly.

---

## 2. Core OOP Concepts & Pillars Used

* **Encapsulation & Composition:** A `ShoppingCart` object contains a collection of `Product` objects.
* **Operator Overloading (Dunder Methods):**
* `__add__` (`+`): Merges two shopping carts into a combined new cart.
* `__sub__` (`-`): Removes a product from the cart by name.
* `__len__`: Returns the total number of items in the cart.
* `__str__`: Provides a clean, readable string representation of the cart.


* **File Handling (Persistence):** Automatically saves and loads cart data to/from a local `shopping_cart.json` file so state is retained across sessions without a database.

---

## 3. Complete Python Source Code (CLI, File Handling & Dunder Methods)

```python
import json
import os

class Product:
    def __init__(self, name, price, quantity=1):
        self.name = name
        self.price = float(price)
        self.quantity = int(quantity)

    def to_dict(self):
        return {"name": self.name, "price": self.price, "quantity": self.quantity}

class ShoppingCart:
    def __init__(self, owner="Guest"):
        self.owner = owner
        self.items = []  # List of Product objects

    # --- Dunder Methods ---
    def __str__(self):
        if not self.items:
            return f"ShoppingCart('{self.owner}') is currently empty."
        
        output = f"\n=== {self.owner.upper()}'S SHOPPING CART ===\n"
        total = 0
        for idx, item in enumerate(self.items, 1):
            subtotal = item.price * item.quantity
            total += subtotal
            output += f"{idx}. {item.name:<15} | ${item.price:>6.2f} x {item.quantity} = ${subtotal:>6.2f}\n"
        output += "-" * 42 + f"\nTotal Cart Value: ${total:.2f}\n"
        return output

    def __len__(self):
        return sum(item.quantity for item in self.items)

    def __add__(self, other):
        """Overloads the '+' operator to merge two shopping carts."""
        if not isinstance(other, ShoppingCart):
            raise TypeError("Can only merge with another ShoppingCart instance.")
        
        new_cart = ShoppingCart(f"{self.owner} & {other.owner}")
        # Copy items from self
        item_map = {}
        for item in self.items:
            item_map[item.name] = Product(item.name, item.price, item.quantity)
        
        # Merge items from other
        for item in other.items:
            if item.name in item_map:
                item_map[item.name].quantity += item.quantity
            else:
                item_map[item.name] = Product(item.name, item.price, item.quantity)
                
        new_cart.items = list(item_map.values())
        return new_cart

    def __sub__(self, product_name):
        """Overloads the '-' operator to remove an item by name."""
        initial_len = len(self.items)
        self.items = [item for item in self.items if item.name.lower() != product_name.lower()]
        if len(self.items) < initial_len:
            print(f"Success: '{product_name}' removed from cart.")
        else:
            print(f"Notice: '{product_name}' was not found in the cart.")
        return self

    # --- File Handling Methods ---
    def save_to_file(self, filename="shopping_cart.json"):
        data = {
            "owner": self.owner,
            "items": [item.to_dict() for item in self.items]
        }
        try:
            with open(filename, "w") as f:
                json.dump(data, f, indent=4)
            print(f"Success: Cart saved to '{filename}'.")
        except IOError as e:
            print(f"Error saving file: {e}")

    def load_from_file(self, filename="shopping_cart.json"):
        if not os.path.exists(filename):
            print(f"Notice: No saved file '{filename}' found.")
            return
        try:
            with open(filename, "r") as f:
                data = json.load(f)
                self.owner = data.get("owner", "Guest")
                self.items = [Product(i["name"], i["price"], i["quantity"]) for i in data.get("items", [])]
            print(f"Success: Cart loaded from '{filename}'.")
        except IOError as e:
            print(f"Error loading file: {e}")

def main():
    cart = ShoppingCart("Alice")
    
    while True:
        print("\n--- SMART SHOPPING CART SYSTEM ---")
        print("1. View Cart & Total (Uses __str__ & __len__)")
        print("2. Add Item to Cart")
        print("3. Remove Item using '-' Operator (__sub__)")
        print("4. Merge with Partner's Cart using '+' Operator (__add__)")
        print("5. Save Cart to File (File Handling)")
        print("6. Load Cart from File (File Handling)")
        print("7. Exit")

        choice = input("Enter your choice (1-7): ").strip()

        if choice == '1':
            print(cart)
            print(f"Total unique/item quantity count in cart: {len(cart)}")

        elif choice == '2':
            name = input("Enter product name: ").strip()
            try:
                price = float(input("Enter unit price ($): "))
                qty = int(input("Enter quantity: "))
                cart.items.append(Product(name, price, qty))
                print(f"Added {qty}x '{name}' to cart.")
            except ValueError:
                print("Error: Invalid number format for price or quantity.")

        elif choice == '3':
            name = input("Enter exact product name to remove: ").strip()
            cart - name  # Invokes __sub__ dunder method

        elif choice == '4':
            print("\n--- Creating a Partner's Cart to Merge ---")
            partner_cart = ShoppingCart("Bob")
            partner_cart.items.append(Product("Wireless Mouse", 25.00, 1))
            partner_cart.items.append(Product("Coffee Beans", 15.50, 2))
            print(partner_cart)
            
            # Using '+' dunder operator
            merged_cart = cart + partner_cart
            print("\nResult of cart1 + cart2:")
            print(merged_cart)
            
            if input("Do you want to adopt this merged cart? (y/n): ").strip().lower() == 'y':
                cart = merged_cart
                print("Cart updated to merged version.")

        elif choice == '5':
            cart.save_to_file()

        elif choice == '6':
            cart.load_from_file()

        elif choice == '7':
            print("Exiting Shopping Cart System. Goodbye!")
            break
        else:
            print("Invalid choice. Please pick between 1 and 7.")

main()
```
