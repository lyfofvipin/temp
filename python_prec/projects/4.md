# Project 4: Ride-Sharing App (RideRequest)

## 1. Real-World Scenario & Theoretical Foundation

Ride-hailing applications like Uber or Lyft coordinate transportation logistics in real-time. The system captures passenger requests, origin/destination points, matched drivers, live trip lifecycle statuses, and distance-based fare calculations. Each active trip is modeled as an instance of the `RideRequest` class, stored and managed within a centralized collection.

---

## 2. Core OOP Concepts & Pillars Used

* **Encapsulation:** Bundling trip data (`rider`, `pickup`, `dropoff`, `driver`, `status`) and trip behaviors (`assign_driver`, `update_trip`, `complete_ride`, `cancel_ride`) inside the `RideRequest` class.
* **State Lifecycle Management:** Tracking dynamic progress states (e.g., `"Searching for driver"`, `"Driver en route"`, `"Completed"`, `"Cancelled"`).
* **CRUD Architecture:**
* **Create:** Request a new ride by submitting passenger details and locations.
* **Read:** View active trip status, assigned driver name, and route details.
* **Update:** Update dropoff location, reassign a driver, or change ride status.
* **Delete:** Cancel/delete a ride request before it is completed.

## 3. Complete Python Source Code (CLI & Core Logic with Full CRUD)

```python
class RideRequest:
    def __init__(self, rider_name, pickup_location, dropoff_location):
        self.rider = rider_name
        self.pickup = pickup_location
        self.dropoff = dropoff_location
        self.driver = None
        self.status = "Searching for driver"

    def assign_driver(self, driver_name):
        self.driver = driver_name
        self.status = "Driver en route"
        print(f"Success: Driver {driver_name} assigned to {self.rider}'s trip.")

    def update_dropoff(self, new_dropoff):
        self.dropoff = new_dropoff
        print(f"Success: Dropoff location updated to '{new_dropoff}'.")

    def complete_ride(self, distance_miles):
        self.status = "Completed"
        fare = 2.50 + (float(distance_miles) * 1.75)  # Base fare + per mile rate
        print(f"Ride finished. Total fare billed to {self.rider}: ${fare:.2f}")
        return fare

    def display_ride_info(self):
        print("\n" + "="*40)
        print("          RIDE REQUEST DETAILS")
        print("="*40)
        print(f"Rider Name : {self.rider}")
        print(f"Pickup     : {self.pickup}")
        print(f"Dropoff    : {self.dropoff}")
        print(f"Driver     : {self.driver if self.driver else 'Not Assigned Yet'}")
        print(f"Status     : {self.status}")
        print("="*40)

def find_ride(rides_list, rider_name):
    for ride in rides_list:
        if ride.rider.lower() == rider_name.lower():
            return ride
    return None

def main():
    rides = []

    while True:
        print("\n--- RIDE-SHARING MANAGEMENT SYSTEM (CRUD) ---")
        print("1. Request New Ride (Create)")
        print("2. Check Ride Status (Read)")
        print("3. Assign Driver / Update Dropoff (Update)")
        print("4. Complete Ride & Calculate Fare (Update)")
        print("5. Cancel/Delete Ride Request (Delete)")
        print("6. Exit")

        choice = input("Enter your choice (1-6): ").strip()

        if choice == '1':
            print("\n--- Request New Ride ---")
            rider = input("Enter Rider Name: ").strip()
            if find_ride(rides, rider):
                print("Error: An active ride request already exists for this rider!")
                continue
            pickup = input("Enter Pickup Location: ").strip()
            dropoff = input("Enter Dropoff Location: ").strip()
            
            new_ride = RideRequest(rider, pickup, dropoff)
            rides.append(new_ride)
            print(f"Success! Ride requested for {rider}. Searching for drivers...")

        elif choice == '2':
            print("\n--- Check Ride Status ---")
            rider = input("Enter Rider Name: ").strip()
            ride = find_ride(rides, rider)
            if ride:
                ride.display_ride_info()
            else:
                print("No active ride found for this rider.")

        elif choice == '3':
            print("\n--- Update Ride Details ---")
            rider = input("Enter Rider Name: ").strip()
            ride = find_ride(rides, rider)
            if ride:
                print("1. Assign Driver")
                print("2. Update Dropoff Location")
                sub_choice = input("Choose update type (1/2): ").strip()
                if sub_choice == '1':
                    driver = input("Enter Driver Name: ").strip()
                    ride.assign_driver(driver)
                elif sub_choice == '2':
                    new_dest = input("Enter new Dropoff location: ").strip()
                    ride.update_dropoff(new_dest)
                else:
                    print("Invalid update option.")
            else:
                print("No active ride found.")

        elif choice == '4':
            print("\n--- Complete Ride ---")
            rider = input("Enter Rider Name: ").strip()
            ride = find_ride(rides, rider)
            if ride:
                try:
                    miles = float(input("Enter distance driven (miles): "))
                    ride.complete_ride(miles)
                except ValueError:
                    print("Invalid distance format.")
            else:
                print("No active ride found.")

        elif choice == '5':
            print("\n--- Cancel/Delete Ride ---")
            rider = input("Enter Rider Name to cancel: ").strip()
            ride = find_ride(rides, rider)
            if ride:
                rides.remove(ride)
                print(f"Ride request for {rider} has been cancelled and deleted.")
            else:
                print("No active ride found.")

        elif choice == '6':
            print("Exiting Ride-Sharing System. Goodbye!")
            break
        else:
            print("Invalid option. Please select between 1 and 6.")

main()
```
