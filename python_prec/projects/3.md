# Project 3: Hotel Room Management System

## 1. Real-World Scenario & Theoretical Foundation

Hotel front-desk software requires robust room inventory management. Receptionists need to inspect room status (vacant vs. occupied), manage guest check-ins and check-outs, update nightly rates, and add newly built rooms to the hotel property catalog. Each room is modeled as an instance of the `HotelRoom` class.

---

## 2. Core OOP Concepts & Pillars Used

* **Encapsulation:** Grouping physical room attributes (`room_number`, `room_type`, `price`, `is_occupied`, `guest_name`) and operational behaviors (`check_in`, `check_out`, `update_room`) inside the `HotelRoom` class.
* **Boolean State Flags:** Toggling `is_occupied` to securely manage room availability and prevent double-booking conflicts.
* **CRUD Architecture:**
* **Create:** Add new rooms to the hotel property inventory.
* **Read:** View room statuses, nightly rates, and assigned guests.
* **Update:** Check in guests, check out guests, or update room pricing/type.
* **Delete:** Remove rooms from the hotel system inventory.



---

## 3. Complete Python Source Code (CLI & Core Logic with Full CRUD)

```python
class HotelRoom:
    def __init__(self, room_number, room_type, price_per_night):
        self.room_number = str(room_number)
        self.room_type = room_type
        self.price = float(price_per_night)
        self.is_occupied = False
        self.guest_name = None

    def check_in(self, guest_name):
        if not self.is_occupied:
            self.is_occupied = True
            self.guest_name = guest_name
            print(f"Success: Room {self.room_number} checked in for guest {guest_name}.")
        else:
            print(f"Error: Room {self.room_number} is already occupied by {self.guest_name}!")

    def check_out(self):
        if self.is_occupied:
            print(f"Checkout Summary: Room {self.room_number} vacated by {self.guest_name}.")
            self.is_occupied = False
            self.guest_name = None
            print(f"Room {self.room_number} is now vacant and flagged for housekeeping.")
        else:
            print(f"Error: Room {self.room_number} is already vacant.")

    def update_room(self, room_type, price):
        self.room_type = room_type
        self.price = float(price)
        print(f"Success: Room {self.room_number} updated.")

    def display_room_status(self):
        status = "Occupied" if self.is_occupied else "Vacant"
        guest_info = f" (Guest: {self.guest_name})" if self.is_occupied else ""
        print(f"Room {self.room_number} [{self.room_type}] - ${self.price:.2f}/night - Status: {status}{guest_info}")

def find_room(rooms_list, room_num):
    for room in rooms_list:
        if room.room_number == room_num:
            return room
    return None

def main():
    rooms = [
        HotelRoom(101, "Standard Single", 99.00),
        HotelRoom(102, "Standard Double", 129.00),
        HotelRoom(201, "King Suite", 249.00)
    ]

    while True:
        print("\n--- HOTEL MANAGEMENT SYSTEM (CRUD) ---")
        print("1. Add New Room to Inventory (Create)")
        print("2. View All Rooms Status (Read)")
        print("3. Check-In Guest / Update Room (Update)")
        print("4. Check-Out Guest (Update)")
        print("5. Delete Room from Inventory (Delete)")
        print("6. Exit")

        choice = input("Enter your choice (1-6): ").strip()

        if choice == '1':
            print("\n--- Add New Room ---")
            r_num = input("Enter Room Number: ").strip()
            if find_room(rooms, r_num):
                print("Error: Room number already exists!")
                continue
            r_type = input("Enter Room Type: ").strip()
            try:
                price = float(input("Enter Price per Night ($): "))
                rooms.append(HotelRoom(r_num, r_type, price))
                print(f"Success! Room {r_num} added to inventory.")
            except ValueError:
                print("Invalid price format.")

        elif choice == '2':
            print("\n--- Current Hotel Inventory ---")
            for r in rooms:
                r.display_room_status()

        elif choice == '3':
            print("\n--- Check-In Guest ---")
            r_num = input("Enter Room Number: ").strip()
            room = find_room(rooms, r_num)
            if room:
                guest = input("Enter Guest Full Name: ").strip()
                room.check_in(guest)
            else:
                print("Error: Room not found.")

        elif choice == '4':
            print("\n--- Check-Out Guest ---")
            r_num = input("Enter Room Number: ").strip()
            room = find_room(rooms, r_num)
            if room:
                room.check_out()
            else:
                print("Error: Room not found.")

        elif choice == '5':
            print("\n--- Delete Room ---")
            r_num = input("Enter Room Number to delete: ").strip()
            room = find_room(rooms, r_num)
            if room:
                if room.is_occupied:
                    print("Error: Cannot delete an occupied room!")
                else:
                    rooms.remove(room)
                    print(f"Room {r_num} removed from inventory.")
            else:
                print("Error: Room not found.")

        elif choice == '6':
            print("Exiting Hotel Management System. Goodbye!")
            break
        else:
            print("Invalid option. Please choose between 1 and 6.")

main()
```
