# Project 10: Personal Expense Tracker & Budget Ledger (File Handling & Dunder Methods)

## 1. Real-World Scenario & Theoretical Foundation

Managing personal finances requires tracking daily expenditures, categorizing costs, analyzing monthly budgets, and persisting financial data locally across sessions. This system models an expense ledger where users can log expenses, combine ledger accounts (e.g., merging personal and shared household expenses), compare spending levels between months using comparison operators, and save/load records automatically using local JSON file handling.

---

## 2. Core OOP Concepts & Pillars Used

* **Encapsulation & Composition:** An `ExpenseLedger` object manages a collection of individual `Expense` items.
* **Operator Overloading (Dunder Methods):**
* `__add__` (`+`): Combines two separate expense ledgers into a single combined ledger.
* `__sub__` (`-`): Removes an expense record by its description.
* `__gt__` (`>`): Compares total spending between two ledger accounts.
* `__str__`: Provides a clean, formatted ledger statement and balance summary.


* **File Handling (Persistence):** Automatically saves and loads ledger data to/from a local `expense_ledger.json` file so financial records are retained without a database.

---

## 3. Complete Python Source Code (CLI, File Handling & Dunder Methods)

```python
import json
import os

class Expense:
    def __init__(self, description, amount, category):
        self.description = description
        self.amount = float(amount)
        self.category = category

    def to_dict(self):
        return {
            "description": self.description,
            "amount": self.amount,
            "category": self.category
        }

class ExpenseLedger:
    def __init__(self, owner="Personal"):
        self.owner = owner
        self.expenses = []  # List of Expense objects

    # --- Dunder Methods ---
    def __str__(self):
        if not self.expenses:
            return f"ExpenseLedger('{self.owner}') has no recorded expenses."
        
        output = f"\n=== {self.owner.upper()}'S EXPENSE LEDGER ===\n"
        total = 0
        for idx, exp in enumerate(self.expenses, 1):
            total += exp.amount
            output += f"{idx}. {exp.description:<18} | ${exp.amount:>7.2f} | [{exp.category}]\n"
        output += "-" * 42 + f"\nTotal Expenditure: ${total:.2f}\n"
        return output

    def __add__(self, other):
        """Overloads the '+' operator to merge two expense ledgers."""
        if not isinstance(other, ExpenseLedger):
            raise TypeError("Can only merge with another ExpenseLedger instance.")
        
        new_ledger = ExpenseLedger(f"{self.owner} & {other.owner}")
        new_ledger.expenses = [Expense(e.description, e.amount, e.category) for e in self.expenses]
        for e in other.expenses:
            new_ledger.expenses.append(Expense(e.description, e.amount, e.category))
        return new_ledger

    def __sub__(self, description):
        """Overloads the '-' operator to remove an expense by its description."""
        initial_len = len(self.expenses)
        self.expenses = [e for e in self.expenses if e.description.lower() != description.lower()]
        if len(self.expenses) < initial_len:
            print(f"Success: Expense '{description}' removed.")
        else:
            print(f"Notice: Expense '{description}' not found.")
        return self

    def __gt__(self, other):
        """Overloads the '>' operator to compare total spending between two ledgers."""
        if not isinstance(other, ExpenseLedger):
            return NotImplemented
        self_total = sum(e.amount for e in self.expenses)
        other_total = sum(e.amount for e in other.expenses)
        return self_total > other_total

    # --- File Handling Methods ---
    def save_to_file(self, filename="expense_ledger.json"):
        data = {
            "owner": self.owner,
            "expenses": [exp.to_dict() for exp in self.expenses]
        }
        try:
            with open(filename, "w") as f:
                json.dump(data, f, indent=4)
            print(f"Success: Ledger saved to '{filename}'.")
        except IOError as e:
            print(f"Error saving file: {e}")

    def load_from_file(self, filename="expense_ledger.json"):
        if not os.path.exists(filename):
            print(f"Notice: No saved file '{filename}' found.")
            return
        try:
            with open(filename, "r") as f:
                data = json.load(f)
                self.owner = data.get("owner", "Personal")
                self.expenses = [
                    Expense(e["description"], e["amount"], e["category"]) 
                    for e in data.get("expenses", [])
                ]
            print(f"Success: Ledger loaded from '{filename}'.")
        except IOError as e:
            print(f"Error loading file: {e}")

def main():
    ledger = ExpenseLedger("Alex")

    while True:
        print("\n--- PERSONAL EXPENSE TRACKER ---")
        print("1. View Ledger & Total (Uses __str__)")
        print("2. Add New Expense")
        print("3. Remove Expense using '-' Operator (__sub__)")
        print("4. Merge with Partner Ledger using '+' Operator (__add__)")
        print("5. Compare Spending using '>' Operator (__gt__)")
        print("6. Save Ledger to File (File Handling)")
        print("7. Load Ledger from File (File Handling)")
        print("8. Exit")

        choice = input("Enter your choice (1-8): ").strip()

        if choice == '1':
            print(ledger)

        elif choice == '2':
            desc = input("Enter expense description (e.g., Grocery): ").strip()
            try:
                amt = float(input("Enter amount ($): "))
                cat = input("Enter category (e.g., Food, Utility, Entertainment): ").strip()
                ledger.expenses.append(Expense(desc, amt, cat))
                print(f"Recorded expense: '{desc}' (${amt:.2f}).")
            except ValueError:
                print("Error: Invalid number format for amount.")

        elif choice == '3':
            desc = input("Enter exact description of expense to remove: ").strip()
            ledger - desc  # Invokes __sub__ dunder method

        elif choice == '4':
            print("\n--- Creating a Partner Ledger to Merge ---")
            partner_ledger = ExpenseLedger("Taylor")
            partner_ledger.expenses.append(Expense("Electricity Bill", 85.50, "Utility"))
            partner_ledger.expenses.append(Expense("Movie Tickets", 32.00, "Entertainment"))
            print(partner_ledger)

            # Using '+' dunder operator
            combined_ledger = ledger + partner_ledger
            print("\nResult of ledger1 + ledger2:")
            print(combined_ledger)

            if input("Adopt this combined ledger? (y/n): ").strip().lower() == 'y':
                ledger = combined_ledger
                print("Ledger updated to combined version.")

        elif choice == '5':
            print("\n--- Compare Spending ---")
            other_ledger = ExpenseLedger("Roommate")
            other_ledger.expenses.append(Expense("Rent", 1200.00, "Housing"))
            print(other_ledger)

            is_higher = ledger > other_ledger
            print(f"Is {ledger.owner}'s spending greater than {other_ledger.owner}'s? {is_higher}")

        elif choice == '6':
            ledger.save_to_file()

        elif choice == '7':
            ledger.load_from_file()

        elif choice == '8':
            print("Exiting Expense Tracker. Goodbye!")
            break
        else:
            print("Invalid choice. Please pick between 1 and 8.")

if __name__ == "__main__":
    main()
```
